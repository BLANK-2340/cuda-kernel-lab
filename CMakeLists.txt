cmake_minimum_required(VERSION 3.18)
project(kernel_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(reference INTERFACE)
target_include_directories(reference INTERFACE include)
add_executable(cpu_demo examples/cpu_demo.cpp)
target_link_libraries(cpu_demo PRIVATE reference)
include(CTest)
if(BUILD_TESTING)
  add_executable(reference_test tests/reference_test.cpp)
  target_link_libraries(reference_test PRIVATE reference)
  add_test(NAME reference COMMAND reference_test)
  add_executable(reduction_test tests/reduction_test.cpp)
  target_link_libraries(reduction_test PRIVATE reference)
  add_test(NAME reduction COMMAND reduction_test)
  add_executable(matmul_test tests/matmul_test.cpp)
  target_link_libraries(matmul_test PRIVATE reference)
  add_test(NAME matmul COMMAND matmul_test)
endif()
option(KERNEL_LAB_CUDA "Build optional CUDA benchmark" OFF)
if(KERNEL_LAB_CUDA)
  enable_language(CUDA)
  add_executable(vector_add_bench cuda/vector_add_bench.cu)
  target_link_libraries(vector_add_bench PRIVATE reference)
  set_target_properties(vector_add_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_executable(reduction_bench cuda/reduction_bench.cu)
  target_link_libraries(reduction_bench PRIVATE reference)
  set_target_properties(reduction_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_executable(tiled_matmul_bench cuda/tiled_matmul_bench.cu)
  target_link_libraries(tiled_matmul_bench PRIVATE reference)
  set_target_properties(tiled_matmul_bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
